cmake_minimum_required(VERSION 3.21)
project(cef_native_validation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(USE_SANDBOX ON CACHE BOOL "Require native CEF sandbox" FORCE)
set(CEF_ROOT "" CACHE PATH "Verified extracted SDK")
list(PREPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
# Never resolve native libraries relative to the caller's working directory.
list(REMOVE_ITEM CEF_LINKER_FLAGS "-Wl,-rpath,.")
set(PROBE_SHARED_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/host.cc")
if(OS_WINDOWS)
  include(windows.cmake)
elseif(OS_MAC)
  include(macos.cmake)
elseif(OS_LINUX)
  function(add_cef_probe_targets)
    add_executable(cef-probe main_linux.cc ${PROBE_SHARED_SOURCES})
    SET_EXECUTABLE_TARGET_PROPERTIES(cef-probe)
    target_include_directories(cef-probe PRIVATE "${CEF_ROOT}")
    target_link_libraries(cef-probe PRIVATE libcef_dll_wrapper "${CEF_ROOT}/Release/libcef.so" ${CEF_STANDARD_LIBS})
    set_target_properties(cef-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Release" BUILD_RPATH "$ORIGIN")
    target_compile_definitions(cef-probe PRIVATE CEF_PROBE_RUNTIME_DIR="${CMAKE_BINARY_DIR}/Release")
    set(CEF_TARGET_OUT_DIR "${CMAKE_BINARY_DIR}/Release")
    list(REMOVE_ITEM CEF_BINARY_FILES chrome-sandbox)
    COPY_FILES(cef-probe "${CEF_BINARY_FILES}" "${CEF_ROOT}/Release" "${CEF_TARGET_OUT_DIR}")
    COPY_FILES(cef-probe "${CEF_RESOURCE_FILES}" "${CEF_ROOT}/Resources" "${CEF_TARGET_OUT_DIR}")
  endfunction()
else()
  message(FATAL_ERROR "Unsupported experiment platform")
endif()
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
add_cef_probe_targets()
